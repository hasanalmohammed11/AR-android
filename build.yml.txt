name: Build APK
on:
  workflow_dispatch:
  push:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      # Optional: use your own Firebase file saved as a repo secret named GOOGLE_SERVICES_JSON
      - name: Use real google-services.json (if secret exists)
        env:
          GS: ${{ secrets.GOOGLE_SERVICES_JSON }}
        run: |
          if [ -n "$GS" ]; then echo "$GS" > app/google-services.json; fi

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 8

      - name: Install Android SDK 27 (uses JDK 17 only for sdkmanager)
        run: |
          export JAVA_HOME="$JAVA_HOME_17_X64"
          yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
          "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-27" "build-tools;27.0.3"
        env:
          SKIP_JDK_VERSION_CHECK: "1"

      - uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: 4.4

      - name: Build
        run: gradle assembleProductionDebug --stacktrace

      - uses: actions/upload-artifact@v4
        with:
          name: app-debug
          path: app/build/outputs/apk/production/debug/*.apk
